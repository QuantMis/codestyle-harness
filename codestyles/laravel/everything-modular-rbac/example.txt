# Laravel + Livewire — everything-modular, role dimension
#
# Delta on everything-modular. Module is axis 1, role is axis 2.
#
#   app/Livewire/<Module>/<Role>/           PascalCase   <- ONLY layer with role folders
#   app/Services/<Module>/                  PascalCase   <- no roles, ever
#   resources/views/livewire/<module>/<role>/  kebab-case
#   routes/<module>.php                     one file per MODULE; role = group inside
#
# Split threshold: > 2 role conditionals in one component or its view.
# Below that: one component, scope the query, @can the buttons.

# routes/complaints.php
#
#   URLs  role-first    /admin/complaints          <- users read URLs
#   NAMES module-first  complaints.admin.index     <- code reads names; must match the
#                                                     folder path Complaints/Admin/
#   This divergence is deliberate. Do not "fix" it.

use App\Livewire\Complaints\{Admin, Staff, Technician};
use App\Livewire\Complaints\ComplaintDetail;
use Illuminate\Support\Facades\Route;

Route::middleware(['auth'])->name('complaints.')->group(function () {

    // unsplit component -> NO role segment, in the URL or the name
    Route::livewire('complaints/{complaint}', ComplaintDetail::class)->name('show');

    Route::prefix('admin/complaints')
        ->middleware('can:manage-complaints')          // coarse gate. ONCE, here.
        ->name('admin.')
        ->group(function () {
            Route::livewire('/', Admin\ListComplaints::class)->name('index');
            Route::livewire('/{complaint}', Admin\ComplaintDetail::class)->name('show');
        });

    Route::prefix('technician/complaints')
        ->middleware('can:resolve-complaints')
        ->name('technician.')
        ->group(function () {
            Route::livewire('/', Technician\ListComplaints::class)->name('index');
        });

    Route::prefix('staff/complaints')
        ->name('staff.')
        ->group(function () {
            Route::livewire('/', Staff\ListComplaints::class)->name('index');
            Route::livewire('/create', Staff\ComplaintForm::class)->name('create');
        });
});

# app/Providers/AppServiceProvider.php — boot()
#
# How a role is STORED (enum | users.role column | permission package) is out of scope

Gate::define('manage-complaints',  fn (User $u) => $u->hasRole('admin'));
Gate::define('resolve-complaints', fn (User $u) => $u->hasAnyRole('admin', 'technician'));


# ─── 5. Admin component ──────────────────────────────────────────────────────

namespace App\Livewire\Complaints\Admin;

use App\Models\Complaint;
use App\Models\User;
use Livewire\Attributes\{Computed, Layout, Url};
use Livewire\Component;
use Livewire\WithPagination;

#[Layout('layouts.admin')]                     // role layout — chrome genuinely differs
class ListComplaints extends Component
{
    use WithPagination;

    #[Url]
    public string $status = '';

    public function assign(Complaint $complaint, int $userId): void
    {
        $this->authorize('assign', $complaint);   // policy guards the ROW.
        //  no Gate::allows('manage-complaints') here — the route group already denied it.

        $complaint->update(['assigned_to' => $userId, 'status' => 'assigned']);

        $this->dispatch('complaint-assigned', id: $complaint->id);
    }

    // data via #[Computed], NOT render(). the mirror still derives the view
    // (everything-modular #9): livewire/complaints/admin/list-complaints.blade.php
    #[Computed]
    public function complaints()
    {
        return Complaint::query()
            ->with(['asset', 'assignee'])
            ->when($this->status, fn ($q) => $q->where('status', $this->status))
            ->latest()
            ->paginate(20);
    }

    #[Computed]
    public function technicians()
    {
        // however you model roles — enum, column, package. out of scope here.
        return User::query()->where('role', 'technician')->orderBy('name')->get();
    }
}


# ─── 6. Technician component — same model, different SCOPE ───────────────────
# The scoping is in the QUERY. Never in the view.

namespace App\Livewire\Complaints\Technician;

use App\Models\Complaint;
use Livewire\Attributes\{Computed, Layout};
use Livewire\Component;
use Livewire\WithPagination;

#[Layout('layouts.app')]
class ListComplaints extends Component
{
    use WithPagination;

    public function resolve(Complaint $complaint, string $note): void
    {
        $this->authorize('resolve', $complaint);

        $complaint->update([
            'status'      => 'resolved',
            'resolution'  => $note,
            'resolved_at' => now(),
        ]);
    }

    #[Computed]
    public function complaints()
    {
        return Complaint::query()
            ->with('asset')
            ->where('assigned_to', auth()->id())        // <- the scope. here, not blade.
            ->whereIn('status', ['assigned', 'in_progress'])
            ->oldest()
            ->paginate(20);
    }
}


# ─── 7. Staff component ──────────────────────────────────────────────────────

namespace App\Livewire\Complaints\Staff;

use App\Models\{Asset, Complaint};
use App\Services\Complaints\ComplaintNumberGenerator;
use Livewire\Attributes\{Computed, Layout, Validate};
use Livewire\Component;

#[Layout('layouts.app')]
class ComplaintForm extends Component                 // NOT SubmitComplaint
{
    #[Validate('required|exists:assets,id')]
    public ?int $asset_id = null;

    #[Validate('required|string|min:10|max:2000')]
    public string $description = '';

    #[Validate('required|in:low,medium,high')]
    public string $priority = 'medium';

    public function save(ComplaintNumberGenerator $numbers): void
    {
        $this->authorize('create', Complaint::class);   // no record yet -> class-level
        $this->validate();

        $complaint = Complaint::create([
            'reference'   => $numbers->next(),        // service knows nothing about roles
            'asset_id'    => $this->asset_id,
            'description' => $this->description,
            'priority'    => $this->priority,
            'status'      => 'open',
            'reported_by' => auth()->id(),            // auth() in the COMPONENT — fine
        ]);

        session()->flash('status', "Complaint {$complaint->reference} submitted.");

        $this->redirectRoute('complaints.staff.index');   // role-segmented name
    }

    #[Computed]
    public function assets()
    {
        return Asset::orderBy('name')->get();
    }
}


# ─── 8. Policy — flat in app/Policies, roles handled INSIDE ──────────────────

namespace App\Policies;

use App\Models\{Complaint, User};

class ComplaintPolicy
{
    public function create(User $user): bool
    {
        return true;                                  // every authenticated role may report
    }

    public function view(User $user, Complaint $complaint): bool
    {
        return $user->hasRole('admin')
            || $complaint->reported_by === $user->id
            || $complaint->assigned_to === $user->id;
    }

    public function assign(User $user, Complaint $complaint): bool
    {
        return $user->hasRole('admin');
    }

    public function resolve(User $user, Complaint $complaint): bool
    {
        return $complaint->assigned_to === $user->id
            && $complaint->status !== 'closed';
    }
}
