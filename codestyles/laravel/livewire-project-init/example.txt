# ─── 3. Delta A — strict types via Pint ──────────────────────────────────────
# pint.json  (kit ships only {"preset": "laravel"})

{
    "preset": "laravel",
    "rules": {
        "declare_strict_types": true
    }
}

# Then normalize the whole tree ONCE, as its own commit:
#   composer lint && git commit -am "chore: enforce strict_types"
# Never hand-type declare(strict_types=1). Never exempt a file.


# ─── 3b. Delta A fallout — REQUIRED, or CI stays red ─────────────────────────
# strict_types kills Stringable->string coercion, so Larastan level 7 now fails
# on the kit's OWN trait:
#
#   app/Concerns/ProfileValidationRules.php:43
#     Method ...::emailRules() should return
#       array<int, array<mixed>|ValidationRule|string>
#     but returns array<int, Unique|string>.
#
# Cause: Rule::unique() returns Illuminate\Validation\Rules\Unique, which
#        implements Stringable — NOT ValidationRule.
# Fix:   name the concrete rule class in the union. The kit's sibling trait
#        PasswordValidationRules already does exactly this with `Password`.

  use Illuminate\Validation\Rule;
+ use Illuminate\Validation\Rules\Unique;

  /**
-  * @return array<int, ValidationRule|array<mixed>|string>
+  * @return array<int, Unique|ValidationRule|array<mixed>|string>
   */
  protected function emailRules(?int $userId = null): array

# Apply to all three annotations in that trait (profileRules, nameRules, emailRules).
# NOT allowed: @phpstan-ignore, a baseline, or dropping phpstan.neon to level 5.


# ─── 4. Delta B — committed pre-commit hook ──────────────────────────────────
# .githooks/pre-commit   (chmod +x)

#!/bin/sh
set -e
composer lint
composer types:check

# Wire it (hooks live in the repo, not in .git/hooks):

#       "@php artisan migrate --force",
#       "@hooks:install",              <-- added

# ─── 5. Gate entrypoints — composer scripts only ─────────────────────────────
# Kit-provided. Use these everywhere (CI, hooks, README). Never vendor/bin/* directly.

composer setup          # install + .env + key + migrate + hooks + npm build
composer dev            # local dev server
composer lint           # pint --parallel            (fix)
composer lint:check     # pint --parallel --test     (verify)
composer types:check    # phpstan analyse
composer test           # config:clear + lint:check + types:check + artisan test
composer ci:check       # what .github/workflows/tests.yml runs


# ─── 6. Routing full-page components ─────────────────────────────────────────
# routes/settings.php — the kit's own pattern; copy it per domain.

Route::middleware(['auth'])->group(function () {
    Route::livewire('settings/profile', Profile::class)->name('profile.edit');
});

# NOT: Route::get('settings/profile', Profile::class)
# routes/web.php stays thin:  require __DIR__.'/settings.php';


# ─── 7. Definition of done ───────────────────────────────────────────────────
# Init is finished when this is green on a clean clone — not when laravel new exits.
